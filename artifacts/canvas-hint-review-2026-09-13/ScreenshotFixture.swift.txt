import SwiftUI
import XCTest
@testable import Steps4


@MainActor
final class CanvasHintReviewCapture: XCTestCase {
    func testCaptureHintSizesAndThemes() async throws {
        let scene = try XCTUnwrap(UIApplication.shared.connectedScenes.first as? UIWindowScene)
        let previous = scene.windows.first { $0.isKeyWindow }
        let cases: [(String, CGFloat, CGFloat, DynamicTypeSize, String, String)] = [
            ("morning-light", 393, 360, .large, "#E4D7AE", "What have you done so far?"),
            ("evening-dark", 393, 360, .large, "#202D3C", "What did you do today?"),
            ("compact", 320, 360, .large, "#E4D7AE", "What have you done so far?"),
            ("large-type", 393, 500, .accessibility3, "#E4D7AE", "What have you done so far?"),
            ("landscape", 852, 360, .large, "#78966B", "What did you do today?")
        ]
        for (name, width, height, type, background, prompt) in cases {
            let palette = CanvasChromePalette.resolve(backgroundColors: [DayObjectRGB(hex: background)])
            let root = VStack {
                Spacer()
                CanvasBottomActionRow(
                    isDataPanelOpen: false, isHappeningPalettePresented: false,
                    soundAppearance: .readyToPlay, addHint: prompt,
                    onSound: {}, onOpenHappeningList: {}, onToggleHappeningPalette: {}
                ).padding(.horizontal, 16).padding(.bottom, 32)
            }
            .frame(width: width, height: height)
            .background(Color(hex: background))
            .environment(\.canvasChromePalette, palette)
            .environment(\.dynamicTypeSize, type)
            let host = UIHostingController(rootView: root)
            let window = UIWindow(windowScene: scene)
            window.frame = CGRect(x: 0, y: 0, width: width, height: height)
            window.rootViewController = host
            window.makeKeyAndVisible()
            host.view.frame = window.bounds
            host.view.layoutIfNeeded()
            try await Task.sleep(for: .milliseconds(250))
            let image = UIGraphicsImageRenderer(bounds: window.bounds).image { _ in
                window.drawHierarchy(in: window.bounds, afterScreenUpdates: true)
            }
            let attachment = XCTAttachment(image: image)
            attachment.name = name
            attachment.lifetime = .keepAlways
            add(attachment)
            window.isHidden = true
        }
        previous?.makeKeyAndVisible()
    }
}
