# Circle and Snowflake Procedural Fills Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make Circle and the current Snowflake body obey the day's five-kind procedural texture policy while preserving legacy gradients, Snowflake trails, deterministic export, and the existing frame budget.

**Architecture:** Keep `ProceduralTexture` as the single geometry/drawing engine. Circle supplies a constant radial profile; `RectMorphFrame` exposes the Snowflake generator's existing radial profile in world-angle order. `RenderCache` keys geometry by family and generates from a canonical bucket time so live, history, thumbnail, and export calls cannot disagree based on call order.

**Tech Stack:** Swift 6, SwiftUI `Canvas`/`Path`/`GraphicsContext`, XCTest, existing `SeededRNG`, `ProceduralTexture`, and `RenderCache`; iOS 17+; no JavaScript, WASM, third-party geometry dependency, or new Metal path.

**Spec:** `docs/superpowers/specs/2026-08-15-circle-snowflake-fills-design.md`

## Global Constraints

- Apply textures to Circle and only the current Snowflake body; never texture Snowflake trail ghosts.
- Rays, legacy `.blob`, and hidden `.spirograph` retain their current visual rendering.
- All five kinds (`flat`, `gradient`, `rings`, `hatch`, `stipple`) remain generated by `ProceduralTexture`.
- Existing Circle `.gradient` and Snowflake `.gradient` rendering must remain byte-for-byte in their renderer-owned compatibility branches.
- Do not add persisted fields, change `CanvasElement` coding keys, or migrate canvas JSON.
- Existing historical Circle/Snowflake elements intentionally gain derived textures.
- Never use `hashValue`, global RNG, wall-clock time, `Set` order, or `Dictionary` order in reproducible geometry.
- Texture geometry is generated only on a `RenderCache` miss, never once per frame and never inside the Snowflake ghost loop.
- Live rendering, thumbnail rendering, wallpaper export, and history rendering use the same CPU/vector path.
- Worst measured frame stays below 55 ms in Debug on the iPhone 17 simulator and at or above 18 fps on the physical target device.
- After visual acceptance, new ribbon/arc and shard/cell families receive a separate spec and plan.

```swift
static func stableSeed(for id: UUID) -> UInt64 {
    let prime: UInt64 = 0x0000_0100_0000_01B3
    var hash: UInt64 = 0xCBF2_9CE4_8422_2325
    var uuid = id.uuid
    withUnsafeBytes(of: &uuid) { bytes in
        for byte in bytes {
            hash ^= UInt64(byte)
            hash &*= prime
        }
    }
    return hash
}
```

```swift
    let kind = composition.texturePolicy.kind(forRank: rank)
```

- [ ] **Step 4: Implement `RadialTextureProfile`**

Add above `TextureGeometry` in `ProceduralTexture.swift`:

```swift
struct RadialTextureProfile: Hashable {
    let center: CGPoint
    let outerRadius: Double
    let radii: [Double]

    init(center: CGPoint, sourceRadii: [CGFloat], rotation: Double) {
        precondition(sourceRadii.count >= 3)
        let outer = Double(sourceRadii.max() ?? 1)
        precondition(outer.isFinite && outer > 0)
        self.center = center
        self.outerRadius = outer

        let count = sourceRadii.count
        self.radii = (0..<count).map { index in
            let worldAngle = Double(index) / Double(count) * 2 * .pi
            var localAngle = (worldAngle - rotation)
                .truncatingRemainder(dividingBy: 2 * .pi)
            if localAngle < 0 { localAngle += 2 * .pi }
            let position = localAngle / (2 * .pi) * Double(count)
            let lower = Int(floor(position)) % count
            let upper = (lower + 1) % count
            let fraction = position - floor(position)
            let value = Double(sourceRadii[lower])
                + (Double(sourceRadii[upper]) - Double(sourceRadii[lower])) * fraction
            return value / outer
        }
    }

    static func circle(
        center: CGPoint,
        radius: Double,
        sampleCount: Int = 48
    ) -> RadialTextureProfile {
        RadialTextureProfile(
            center: center,
            sourceRadii: [CGFloat](repeating: CGFloat(radius), count: sampleCount),
            rotation: 0)
    }
}
```

```swift
func textureGeometry(
    family: TextureGeometryFamily,
    seed: UInt64,
    spec: TextureSpec,
    profileKey: Int,
    time: Double,
    radiiAtCanonicalTime: (Double) -> [Double]
) -> TextureGeometry {
    let phase = Self.texturePhase(for: seed)
    let bucket = Self.textureBucket(for: time + phase)
    let key = TextureCacheKey(
        family: family, seed: seed, spec: spec,
        profileKey: profileKey, timeBucket: bucket)
    if let hit = textureCache[key] { return hit }

    if bucket > textureCacheLastPruneBucket {
        textureCache = textureCache.filter { $0.key.timeBucket >= bucket - 1 }
        textureCacheLastPruneBucket = bucket
    }

    let canonicalTime = (Double(bucket) + 0.5) * Self.textureBucketSeconds - phase
    let geometry = ProceduralTexture.geometry(
        spec: spec, radii: radiiAtCanonicalTime(canonicalTime), seed: seed)
    textureCache[key] = geometry
    return geometry
}
```

```swift
@MainActor
final class TempFillPerformanceTests: XCTestCase {
    private let dayKey = "2026-08-10"

    private func elements(_ shape: CanvasShapeType) -> [CanvasElement] {
        let day = DayComposition.forDay(dayKey: dayKey, happeningCount: 15)
        var result: [CanvasElement] = []
        for rank in 0..<15 {
            result.append(CanvasElement.spawn(
                optionId: "perf_\(rank)", label: "P\(rank)",
                existingElements: result, allowedShapeTypes: [shape],
                dayKey: dayKey, composition: day))
        }
        return result
    }

    func testGeometryGenerationCostByFamilyAndKind() {
        let snowflake = ProceduralShapeGenerator.rectMorphFrame(
            seed: 42, time: 20,
            in: CGRect(x: 0, y: 0, width: 200, height: 200))
        let profiles: [(String, [Double])] = [
            ("circle", [Double](repeating: 1, count: 48)),
            ("snowflake", snowflake.textureProfile.radii),
        ]

        for (family, radii) in profiles {
            for kind in TextureKind.allCases {
                let spec = TextureSpec(
                    kind: kind, density: 1, uniformity: 0, angle: 1.2)
                let start = CFAbsoluteTimeGetCurrent()
                var sink = 0
                for seed in 0..<15 {
                    let geometry = ProceduralTexture.geometry(
                        spec: spec, radii: radii, seed: UInt64(seed))
                    sink += geometry.rings.count
                        + geometry.lines.count + geometry.dots.count
                }
                let milliseconds = (CFAbsoluteTimeGetCurrent() - start) * 1_000
                print("FILL_GENERATION family=\(family) kind=\(kind.rawValue) ms=\(milliseconds) sink=\(sink)")
                if kind != .stipple {
                    XCTAssertLessThan(milliseconds, 55)
                }
            }
        }
    }
}
```

```swift
func testCanvasDrawCostByFamilyAndKind() {
    for shape in [CanvasShapeType.circle, .snowflake] {
        let items = elements(shape)
        for kind in TextureKind.allCases {
            let cache = RenderCache()
            let spec = TextureSpec(kind: kind, density: 1, uniformity: 0, angle: 1.2)
            let view = Canvas { context, size in
                for element in items {
                    switch shape {
                    case .circle:
                        CircleShapeRenderer.draw(
                            element, context: &context, size: size, t: 20,
                            decay: 0, blendMode: .normal, ampScale: 1,
                            interaction: nil, decayedColor: .red,
                            decayedColor2: .blue, spec: spec, cache: cache)
                    case .snowflake:
                        SnowflakeShapeRenderer.draw(
                            element, context: &context, size: size, t: 20,
                            decay: 0, blendMode: .normal, ampScale: 1,
                            renderCache: cache, decayedColor: .red,
                            decayedColor2: .blue, spec: spec)
                    default:
                        XCTFail("Unexpected performance-test shape")
                    }
                }
            }.frame(width: 390, height: 844)

            _ = ImageRenderer(content: view).uiImage
            var samples: [Double] = []
            for _ in 0..<5 {
                let start = CFAbsoluteTimeGetCurrent()
                _ = ImageRenderer(content: view).uiImage
                samples.append((CFAbsoluteTimeGetCurrent() - start) * 1_000)
            }
            let average = samples.reduce(0, +) / Double(samples.count)
            let maximum = samples.max() ?? .infinity
            print("FILL_DRAW shape=\(shape.rawValue) kind=\(kind.rawValue) avg=\(average) max=\(maximum)")
            XCTAssertLessThan(maximum, 55)
        }
    }
}
```
